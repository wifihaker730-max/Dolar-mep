  <script>
    // ID de tu Google Sheet
    const SHEET_ID = '1WkHlfGAjluwEBDzw2PkGqdvVmT5vD7rcY-5RJXfJLeE';
    const CSV_URL = `https://docs.google.com/spreadsheets/d/${SHEET_ID}/gviz/tq?tqx=out:csv`;

    let myChart = null;

    async function loadData() {
      try {
        const response = await fetch(CSV_URL);
        const dataText = await response.text();
        
        // Convertir CSV a filas
        const rows = dataText.split('\n').map(row => row.split(',').map(cell => cell.replace(/"/g, '').trim()));
        
        // Omitir encabezado y filtrar filas vacías
        const cleanRows = rows.slice(1).filter(r => r.length >= 3 && r[0] !== "");

        if (cleanRows.length === 0) return;

        // Extraer vectores de datos
        const dates = cleanRows.map(r => r[0]);
        const compras = cleanRows.map(r => parseFloat(r[1].replace(',', '.')) || 0);
        const ventas = cleanRows.map(r => parseFloat(r[2].replace(',', '.')) || 0);

        // Últimos valores
        const lastIdx = cleanRows.length - 1;
        const lastDateStr = dates[lastIdx];
        const lastCompra = compras[lastIdx];
        const lastVenta = ventas[lastIdx];
        const spread = (lastVenta - lastCompra).toFixed(2);

        // Actualizar Tarjetas
        document.getElementById('valCompra').innerText = `$ ${lastCompra.toLocaleString('es-AR', {minimumFractionDigits: 2})}`;
        document.getElementById('valVenta').innerText = `$ ${lastVenta.toLocaleString('es-AR', {minimumFractionDigits: 2})}`;
        document.getElementById('dateCompra').innerText = `Fecha: ${lastDateStr}`;
        document.getElementById('dateVenta').innerText = `Fecha: ${lastDateStr}`;
        document.getElementById('valSpread').innerText = `$ ${spread}`;

        // Evaluar estado (Si la última fecha es de hoy)
        const todayStr = new Date().toLocaleDateString('es-AR');
        const isOnline = (lastDateStr === todayStr);
        
        setSystemStatus(isOnline);

        // Renderizar o actualizar gráfico
        renderChart(dates.slice(-10), compras.slice(-10), ventas.slice(-10));

      } catch (error) {
        console.error("Error al cargar datos:", error);
        setSystemStatus(false);
      }
    }

    function setSystemStatus(isOnline) {
      const dot = document.getElementById('statusDot');
      const text = document.getElementById('statusText');
      if (isOnline) {
        dot.className = 'dot active';
        text.innerText = 'En vivo';
      } else {
        dot.className = 'dot inactive';
        text.innerText = 'Desactivado';
      }
    }

    function renderChart(labels, dataCompra, dataVenta) {
      const ctx = document.getElementById('chartMEP').getContext('2d');
      
      if (myChart) {
        myChart.destroy();
      }

      myChart = new Chart(ctx, {
        type: 'line',
        data: {
          labels: labels,
          datasets: [
            {
              label: 'Dólar MEP Compra',
              data: dataCompra,
              borderColor: '#3182ce',
              backgroundColor: 'rgba(49, 130, 206, 0.1)',
              fill: true,
              tension: 0.3
            },
            {
              label: 'Dólar MEP Venta',
              data: dataVenta,
              borderColor: '#dd6b20',
              backgroundColor: 'rgba(221, 107, 32, 0.1)',
              fill: true,
              tension: 0.3
            }
          ]
        },
        options: {
          responsive: true,
          plugins: {
            legend: { position: 'top' }
          },
          scales: {
            y: { beginAtZero: false }
          }
        }
      });
    }

    // Cargar datos al iniciar y actualizar cada 1 minuto
    loadData();
    setInterval(loadData, 60000);
